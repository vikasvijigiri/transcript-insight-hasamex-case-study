### Task 1: Unit test the eval harness's scoring logic

**Files:**
- Create: `backend/tests/test_eval_scoring.py`

**Interfaces:**
- Consumes: `evals.run_eval.run_case`, `evals.golden_cases.EvalCase` (both already exist, read in full while writing this plan).
- Produces: regression coverage for `run_case`'s pass/fail logic — no change to `run_case` itself, since it's already correct (verified by reading it).

- [ ] **Step 1: Write the tests**

```python
"""Unit tests for the eval harness's own scoring logic (evals/run_eval.py).
No live API calls — a fake object standing in for a Provider is passed
directly to run_case, same no-live-call discipline as the main suite."""

from evals.golden_cases import EvalCase
from evals.run_eval import run_case
from app.providers.types import AskResult, RawCitation
from app.experts import EXPERTS
from app.transcript_parser import load_transcript


class _FakeProvider:
    def __init__(self, result: AskResult):
        self._result = result

    def ask(self, docs, prompt, max_tokens=1024):
        return self._result


def test_golden_case_passes_when_grounded_with_a_real_citation():
    quote = "Very important. The clinical argument may get surgeons interested"
    france = load_transcript("france", EXPERTS[0]["file"])
    offset = france.raw_text.index(quote)
    case = EvalCase(id="t1", expert_id="france", question="q", expect_grounded=True)
    provider = _FakeProvider(
        AskResult(
            answer_text="answer",
            raw_citations=[RawCitation(document_index=0, cited_text=quote, start_char_index=offset)],
        )
    )
    result = run_case(provider, case)
    assert result["passed"] is True
    assert result["is_grounded"] is True
    assert result["fabricated_citations"] == 0


def test_golden_case_fails_if_the_only_citation_is_fabricated():
    case = EvalCase(id="t2", expert_id="france", question="q", expect_grounded=True)
    provider = _FakeProvider(
        AskResult(
            answer_text="answer",
            raw_citations=[RawCitation(document_index=0, cited_text="never said this", start_char_index=0)],
        )
    )
    result = run_case(provider, case)
    assert result["passed"] is False
    assert result["fabricated_citations"] == 1


def test_trap_case_passes_only_with_zero_citations():
    case = EvalCase(id="t3", expert_id="france", question="q", expect_grounded=False)
    provider = _FakeProvider(AskResult(answer_text="Not addressed in the transcript.", raw_citations=[]))
    result = run_case(provider, case)
    assert result["passed"] is True
    assert result["is_grounded"] is False


def test_trap_case_fails_if_it_fabricates_a_citation_anyway():
    case = EvalCase(id="t4", expert_id="france", question="q", expect_grounded=False)
    provider = _FakeProvider(
        AskResult(
            answer_text="Actually, the price is...",
            raw_citations=[RawCitation(document_index=0, cited_text="invented quote", start_char_index=0)],
        )
    )
    result = run_case(provider, case)
    assert result["passed"] is False
```

- [ ] **Step 2: Run to verify current behavior (should already pass — this locks in existing correct behavior, it's not fixing a bug)**

Run: `.venv/Scripts/python.exe -m pytest tests/test_eval_scoring.py -v` (or `uv run pytest tests/test_eval_scoring.py -v` once Phase 1's uv migration has landed)
Expected: all 4 PASS immediately, since `run_case`'s logic was already correct — this task adds coverage, it doesn't change behavior.

Run: `.venv/Scripts/python.exe -m pytest -q`
Expected: total test count increases by 4 over the previous full-suite run.


```ts
import { test, expect } from "@playwright/test";

test("switching experts shows different content", async ({ page }) => {
  await page.goto("/");
  const firstExpertAnswer = page.locator("main").first();
  await page.getByRole("button", { name: /Dr\. Jean Martin/ }).click();
  const franceText = await firstExpertAnswer.textContent();
  await page.getByRole("button", { name: /Anna Keller/ }).click();
  await expect(firstExpertAnswer).not.toHaveText(franceText ?? "");
});

test("asking a question shows an answer with a citation", async ({ page }) => {
  await page.goto("/");
  await page.getByRole("button", { name: /Ask a Question/ }).click();
  await page.getByRole("textbox").fill("What are the main adoption barriers?");
  await page.getByRole("button", { name: /submit|ask/i }).click();
  await expect(page.getByText(/\d{2}:\d{2}/).first()).toBeVisible({ timeout: 15_000 });
});

test("clicking a citation highlights it in the source panel", async ({ page }) => {
  await page.goto("/");
  await page.getByText(/\d{2}:\d{2}/).first().click();
  await expect(page.locator("mark")).toBeVisible();
});

test("themes tab loads common themes and disagreements", async ({ page }) => {
  await page.goto("/");
  await page.getByRole("button", { name: /Themes & Disagreements/ }).click();
  await expect(page.getByText(/Common Themes/i)).toBeVisible({ timeout: 15_000 });
});

test("keyboard-only navigation reaches the skip link and main content", async ({ page }) => {
  await page.goto("/");
  await page.keyboard.press("Tab");
  await expect(page.getByRole("link", { name: /skip to content/i })).toBeFocused();
});
```

Note: these selectors assume the exact button/role text from Phase 2's implementation — if an executor's actual copy differs slightly (e.g. "Submit" vs "Ask"), adjust the selector to match what was actually built, don't change the app's copy to match a guessed selector.

- [ ] **Step 2: Write the scan**

`frontend/e2e/accessibility.spec.ts`:
```ts
import { test, expect } from "@playwright/test";
import AxeBuilder from "@axe-core/playwright";

const screens: { name: string; setup: (page: import("@playwright/test").Page) => Promise<void> }[] = [
  { name: "Expert Q&A", setup: async (page) => { await page.goto("/"); } },
  {
    name: "Themes",
    setup: async (page) => {
      await page.goto("/");
      await page.getByRole("button", { name: /Themes & Disagreements/ }).click();
    },
  },
  {
    name: "Ask a Question",
    setup: async (page) => {
      await page.goto("/");
      await page.getByRole("button", { name: /Ask a Question/ }).click();
    },
  },
];

for (const screen of screens) {
  test(`${screen.name} has no automatically-detectable a11y violations`, async ({ page }) => {
    await screen.setup(page);
    const results = await new AxeBuilder({ page }).withTags(["wcag2a", "wcag2aa", "wcag22aa"]).analyze();
    expect(results.violations, JSON.stringify(results.violations, null, 2)).toEqual([]);
  });
}
```

- [ ] **Step 3: Run it and fix anything it finds**

Run: `npm run test:e2e -- accessibility`
Expected: initially may FAIL with real findings (contrast, missing labels, etc.) — if it does, fix the actual component (not the test) per the violation's specific guidance, then re-run until clean. This step is intentionally open-ended: the whole point of an automated scan is finding things the plan's author didn't anticipate.

**Scoping note on visual regression**: the spec (§10) names Playwright's `toHaveScreenshot()` as the current-practice technique for pixel-diff visual regression. This task captures documentation screenshots instead — a deliberate scope choice, not an oversight: `toHaveScreenshot()` baselines need ongoing maintenance (re-approving every intentional future visual change), which has real value for a long-lived maintained product and little for a one-time case-study submission. If this app continues past submission, adding `toHaveScreenshot()` assertions to Task 3's E2E suite, using this task's screenshots as the initial baselines, is the natural next step — not built now, per the same YAGNI reasoning applied to resizable panels and blanket mutation testing elsewhere in this plan.
